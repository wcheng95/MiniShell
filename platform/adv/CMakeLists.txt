cmake_minimum_required(VERSION 3.16)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(minishell_adv)

# T017: retain the registry component and compile only its RX-loss notification
# patch from the build directory. Hash validation fails closed on source drift.
idf_component_get_property(uac_lib espressif__usb_host_uac COMPONENT_LIB)
idf_component_get_property(uac_dir espressif__usb_host_uac COMPONENT_DIR)
set(uac_patch "${CMAKE_CURRENT_SOURCE_DIR}/patch_uac_rx.py")
set(uac_original "${uac_dir}/uac_host.c")
set(uac_patched "${CMAKE_BINARY_DIR}/t017_uac/uac_host.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${uac_patch}" "${uac_original}")
execute_process(COMMAND "${PYTHON}" "${uac_patch}" "${uac_original}" "${uac_patched}"
    RESULT_VARIABLE patch_result)
if(NOT patch_result EQUAL 0)
    message(FATAL_ERROR "T017 UAC RX loss patch failed")
endif()
get_target_property(uac_sources ${uac_lib} SOURCES)
set(uac_replaced FALSE)
set(uac_build_sources "")
foreach(source IN LISTS uac_sources)
    get_filename_component(source_name "${source}" NAME)
    if(source_name STREQUAL "uac_host.c")
        list(APPEND uac_build_sources "${uac_patched}")
        set(uac_replaced TRUE)
    else()
        list(APPEND uac_build_sources "${source}")
    endif()
endforeach()
if(NOT uac_replaced)
    message(FATAL_ERROR "T017 UAC component source layout changed")
endif()
set_property(TARGET ${uac_lib} PROPERTY SOURCES "${uac_build_sources}")
message(STATUS "T017: UAC 1.3.3 RX losses report TRANSFER_ERROR (build-local patch)")

# I001 USB curiosity diagnostic: replace only ESP-IDF's DWC HCD source with a
# build-local instrumented copy. The ISR patch records integer state only when
# an ISO descriptor is NOT_EXECUTED; reporting remains outside the ISR.
idf_component_get_property(usb_lib usb COMPONENT_LIB)
idf_component_get_property(usb_dir usb COMPONENT_DIR)
set(hcd_diag_patch "${CMAKE_CURRENT_SOURCE_DIR}/patch_hcd_dwc_diag.py")
set(hcd_original "${usb_dir}/hcd_dwc.c")
set(hcd_patched "${CMAKE_BINARY_DIR}/i001_hcd/hcd_dwc.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${hcd_diag_patch}" "${hcd_original}")
execute_process(COMMAND "${PYTHON}" "${hcd_diag_patch}" "${hcd_original}" "${hcd_patched}"
    RESULT_VARIABLE hcd_patch_result)
if(NOT hcd_patch_result EQUAL 0)
    message(FATAL_ERROR "I001 DWC HCD diagnostic patch failed")
endif()
get_target_property(usb_sources ${usb_lib} SOURCES)
set(hcd_replaced FALSE)
set(usb_build_sources "")
foreach(source IN LISTS usb_sources)
    get_filename_component(source_name "${source}" NAME)
    if(source_name STREQUAL "hcd_dwc.c")
        list(APPEND usb_build_sources "${hcd_patched}")
        set(hcd_replaced TRUE)
    else()
        list(APPEND usb_build_sources "${source}")
    endif()
endforeach()
if(NOT hcd_replaced)
    message(FATAL_ERROR "I001 ESP-IDF USB source layout changed")
endif()
set_property(TARGET ${usb_lib} PROPERTY SOURCES "${usb_build_sources}")
message(STATUS "I001: DWC HCD ISO NOT_EXECUTED diagnostics enabled (build-local patch)")
